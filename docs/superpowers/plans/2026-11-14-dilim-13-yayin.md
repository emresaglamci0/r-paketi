```r
test_that("DESCRIPTION meets GitHub release basics", {
  skip_on_cran()
  d <- read.dcf(system.file("DESCRIPTION", package = "plansize"))
  expect_true(all(c("URL", "BugReports", "VignetteBuilder") %in% colnames(d)))
  # full-name check of Authors@R (given and family) moves to the CRAN stage
  expect_lt(nchar(d[, "Title"]), 65)
  expect_false(grepl("^plansize", d[, "Title"], ignore.case = TRUE))
})
test_that("every export is in the pkgdown reference", {
  skip_on_cran(); skip_if_not(file.exists(file.path("..", "..", "_pkgdown.yml")))
  y <- paste(readLines(file.path("..", "..", "_pkgdown.yml")), collapse = "\n")
  for (f in getNamespaceExports("plansize")) expect_true(grepl(f, y, fixed = TRUE), info = f)
})
```
